cmake_minimum_required(VERSION 3.16)
project(lte_scheduler_simulator VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined-behavior checks" OFF)
add_library(lte_core src/simulator.cpp)
target_include_directories(lte_core PUBLIC include)
if(MSVC)
    target_compile_options(lte_core PUBLIC /W4 /WX)
else()
    target_compile_options(lte_core PUBLIC -Wall -Wextra -Wpedantic -Werror)
    if(ENABLE_SANITIZERS)
        target_compile_options(lte_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(lte_core PUBLIC -fsanitize=address,undefined)
    endif()
endif()
add_executable(lte_simulator src/main.cpp)
target_link_libraries(lte_simulator PRIVATE lte_core)
add_executable(unit_tests tests/unit_tests.cpp)
target_link_libraries(unit_tests PRIVATE lte_core)
enable_testing()
foreach(case_name invalid_config cqi_mapping empty_queues queue_capacity invalid_enqueue atomic_input_validation
        outage_skipped round_robin_rotation pf_channel_preference pf_history_changes_selection resource_budget
        segmentation_and_latency fifo_latency idle_ewma metric_units equal_channels_fairness deterministic_replay randomized_conservation)
    add_test(NAME ${case_name} COMMAND unit_tests ${case_name})
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME cli_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_cli.py $<TARGET_FILE:lte_simulator>)
